```javascript
function addStockTypeItem() {
  const lInp = document.getElementById('new-stock-type-input');
  const uInp = document.getElementById('new-stock-unit-input');
  const l = lInp.value.trim(); const u = uInp.value.trim() || 'adet';
  if (!l) return;
  const v = l.toLowerCase().replace(/\s+/g,'_').replace(/[^a-z0-9_]/g,'');
  const list = getStockTypes(); list.push({v, l, unit: u});
  settings.stockTypes = list; lInp.value = ''; uInp.value = '';
  saveSettings(); renderSettingLists(); showToast('Stok türü eklendi');
}
function moveStockType(i, dir) {
  const list = getStockTypes(); const j = i + dir;
  if (j < 0 || j >= list.length) return;
  [list[i], list[j]] = [list[j], list[i]];
  settings.stockTypes = list; saveSettings(); renderSettingLists();
}
function deleteStockType(i) {
  const list = getStockTypes(); list.splice(i, 1);
  settings.stockTypes = list; saveSettings(); renderSettingLists(); showToast('Stok türü silindi');
}
```

```javascript
function createNewSeason() {
  const year = document.getElementById('new-season-year').value;
  const name = document.getElementById('new-season-name').value.trim();
  if (!year || !name) { showToast('Yıl ve ad zorunlu'); return; }
  const newSeasonId = 'season_' + genId();
  const s = { id: newSeasonId, year: Number(year), name, active: false };
  data.seasons.push(s);
  
  // Kovan aktarımı: kovanların season kaydı yok ama hiveLog için newSeason bağlantısı eklenebilir
  // Şimdilik: kovanlar zaten tüm sezonlarda görünür (season filtresi yok), bu bilgi gösterimsel
  const transferHives = document.getElementById('transfer-hives')?.checked;
  const transferStock = document.getElementById('transfer-stock')?.checked;
  const transferTasks = document.getElementById('transfer-tasks')?.checked;
  
  let notes = [];
  if (transferHives) notes.push(data.hives.length + ' kovan');
  if (transferStock) {
    // Mevcut stok bakiyelerini yeni sezona "devir" olarak kaydet
    const stockProducts = [...new Set(data.stock.map(s => s.product))];
    stockProducts.forEach(product => {
      const balance = data.stock.filter(s => s.product === product)
        .reduce((sum, s) => s.action === 'in' ? sum + Number(s.amount) : sum - Number(s.amount), 0);
      if (balance > 0) {
        data.stock.push({ id: genId(), product, action: 'in', amount: balance, unit: 'kg', date: year + '-01-01', note: (Number(year)-1) + ' sezonu devri' });
      }
    });
    notes.push('stok devri');
  }
  if (transferTasks) {
    const pendingTasks = data.tasks.filter(t => !t.done);
    // Görevler zaten data.tasks'ta, aktarım = var olduklarını not et
    notes.push(pendingTasks.length + ' bekleyen görev');
  }
  
  saveData();
  closeModal('new-season-modal');
  renderSeasons(); renderSeasonsInSettings();
  showToast('Sezon oluşturuldu' + (notes.length ? ' (' + notes.join(', ') + ' aktarıldı)' : ''));
}
```

```javascript
let calCurrentMonth = new Date();
let calSelectedDate = null;

function renderMiniCal() {
  const el = document.getElementById('cal-mini');
  if (!el) return;
  const y = calCurrentMonth.getFullYear();
  const m = calCurrentMonth.getMonth();
  const today = new Date(); today.setHours(0,0,0,0);
  const firstDay = new Date(y, m, 1);
  const lastDay = new Date(y, m+1, 0);
  const startDow = (firstDay.getDay() + 6) % 7; // Pazartesi=0
  
  const taskDates = new Set(data.tasks.filter(t => t.dueDate).map(t => t.dueDate));
  
  const monthNames = ['Ocak','Şubat','Mart','Nisan','Mayıs','Haziran','Temmuz','Ağustos','Eylül','Ekim','Kasım','Aralık'];
  const dayNames = ['Pt','Sa','Ça','Pe','Cu','Ct','Pz'];
  
  let days = '';
  // Boş başlangıç hücreleri
  for (let i = 0; i < startDow; i++) days += `<div class="cal-day other-month"></div>`;
  // Ayın günleri
  for (let d = 1; d <= lastDay.getDate(); d++) {
    const dateStr = `${y}-${String(m+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
    const isToday = new Date(y,m,d).getTime() === today.getTime();
    const isSel = calSelectedDate === dateStr;
    const hasTsk = taskDates.has(dateStr);
    days += `<div class="cal-day ${isToday?'today':''} ${isSel?'selected':''} ${hasTsk?'has-task':''}" onclick="selectCalDay('${dateStr}')">${d}</div>`;
  }
  
  el.innerHTML = `
    <div class="cal-header">
```

```javascript
function renderCalTaskList() {
  const el = document.getElementById('cal-task-list');
  if (!el) return;
  const calTasks = data.tasks.filter(t => t.dueDate);
  if (!calTasks.length) { el.innerHTML = '<div style="font-size:12px;color:var(--text-soft);padding:4px 0">Takvimli görev yok. Takvimde bir güne tıklayın.</div>'; return; }
  
  const today = today_str();
  const grouped = {};
  calTasks.forEach(t => { if (!grouped[t.dueDate]) grouped[t.dueDate] = []; grouped[t.dueDate].push(t); });
  const sortedDates = Object.keys(grouped).sort();
  
  el.innerHTML = sortedDates.map(date => {
    const isOverdue = date < today;
    const isToday = date === today;
    const label = isToday ? 'Bugün' : isOverdue ? `⚠️ ${fmtDate(date)}` : fmtDate(date);
    const color = isOverdue ? 'var(--red-warn)' : isToday ? 'var(--honey-dark)' : 'var(--text-soft)';
    const tasks = grouped[date].map(t => `
      <div class="task-item">
        <div class="task-checkbox ${t.done?'checked':''}" onclick="toggleTask('${t.id}')">${t.done?'✓':''}</div>
        <div class="task-text ${t.done?'done':''}">${t.text}</div>
        <button onclick="deleteTask('${t.id}')" style="background:none;border:none;font-size:16px;color:#CCC;cursor:pointer">✕</button>
      </div>
    `).join('');
    return `<div style="margin-bottom:10px"><div style="font-size:11px;font-weight:800;color:${color};margin-bottom:4px">${label}</div>${tasks}</div>`;
  }).join('');
}

function today_str() { return new Date().toISOString().split('T')[0]; }
```
